이번 시간에는 **wxPython**의 **다이얼로그(Dialog)**에 대해 배워 볼 건데요.

**다이얼로그(Dialog)**는 사용자로부터 정보를 입력받거나, 메시지를 표시하거나, 설정을 변경하는 등 애플리케이션과 사용자가 상호작용하는 별도의 **팝업창(대화상자)**을 의미해요.
메인 창과는 분리되어 작동하며, 주로 OK/Cancel 버튼과 함께 사용됩니다.

## 미리 정의된 다이얼로그

미리 정의되어 있는 **다이얼로그(Dialog)**에 대해 배워 볼 건데요.
GUI의 메뉴를 통해 여러 다이얼로그를 실행하는 예제를 만들어 보면서 간단하게 **메뉴바(MenuBar)**와 **상태바(StatusBar)**도 만들어 볼 거에요.

```python
# -*- coding: utf-8 -*-
# common_dialogs.py

import wx
import os, sys

class MyFrame(wx.Frame):
	def __init__(self, parent, title: str):
		super().__init__(parent, title=title)

		# 상태바 생성
		self.CreateStatusBar()

		# 메뉴바 생성
		mb = wx.MenuBar()
		# 메뉴 추가
		menu = wx.Menu()
		menu.Append(99, '&Message Dialog', 'Shows a Message Dialog')
		menu.Append(100, '&Color Dialog', 'Shows a Color Dialog')
		menu.Append(101, '&File Dialog', 'Shows a File Dialog')
		menu.Append(102, '&Page Setup Dialog', 'Shows a Page Setup Dialog')
		menu.Append(103, '&Font Dialog', 'Shows a Font Dialog')
		menu.Append(104, '&Directory Dialog', 'Shows a Directory Dialog')
		menu.Append(105, '&SingleChoice Dialog', 'Shows a SingleChoice Dialog')
		menu.Append(106, '&TextEntry Dialog', 'Shows a TextEntry Dialog')
		# 메뉴별 이벤트 함수 연결
		item = menu.Append(-1, '&RearrangeDialog', 'Show a Rearrange Dialog')
		menu.Bind(wx.EVT_MENU, self.on_rearrange, item)
		mb.Append(menu, '&Dialogs')
		# 프레임에 메뉴 셋팅
		self.SetMenuBar(mb)

		# 메뉴별 이벤트 함수 연결
		self.Bind(wx.EVT_MENU, self.on_message, id=99)
		self.Bind(wx.EVT_MENU, self.on_choosecolor, id=100)
		self.Bind(wx.EVT_MENU, self.on_openfile, id=101)
		self.Bind(wx.EVT_MENU, self.on_pagesetup, id=102)
		self.Bind(wx.EVT_MENU, self.on_choosefont, id=103)
		self.Bind(wx.EVT_MENU, self.on_opendir, id=104)
		self.Bind(wx.EVT_MENU, self.on_singlechoice, id=105)
		self.Bind(wx.EVT_MENU, self.on_textentry, id=106)

		# 패널 생성
		panel = wx.Panel(self)

	# 메세지 다이얼로그 이벤트
	def on_message(self, event):
		dlg = wx.MessageDialog(self, '한 생명을 구하는 것은 세상을 구한 것과 같다.', '탈무드', wx.OK|wx.ICON_INFORMATION)
		dlg.SetOKLabel('계속')
		dlg.ShowModal()
		dlg.Destroy()

	# 색상 다이얼로그 이벤트
	def on_choosecolor(self, event):
		dlg = wx.ColourDialog(self)
		# 풀 색상 선택창 사용 유무 지정(True: 사용 / False: 미사용)
		dlg.GetColourData().SetChooseFull(False)
		# 색상 다이얼로그에서 색상을 선택후 Ok 버튼을 누르면
		if dlg.ShowModal() == wx.ID_OK:
			# 선택한 색상값을 가져옴
			data = dlg.GetColourData()
			# 선택한 색상 값을 Frame의 상태바에 출력함
			self.SetStatusText('Selected Color: {}'.format(str(data.GetColour().Get())))
		# 색상 다이얼로그 파괴
		dlg.Destroy()

	# 파일 오픈 다이얼로그 이벤트
	def on_openfile(self, event):
		dirname = os.getcwd()
		dlg = wx.FileDialog(self, '열기', defaultDir=dirname, defaultFile='', wildcard='텍스트 문서 (*.txt)|*.txt|모든 파일 (*.*)|*.*', style=wx.FD_OPEN)
		# 파일 다이얼로그에서 파일을 선택후 Ok 버튼을 누르면
		if dlg.ShowModal() == wx.ID_OK:
			# 파일의 경로를 구함
			path = dlg.GetPath()
			filename = dlg.GetFilename()
			# 선택한 파일의 이름을 Frame의 상태바에 출력함
			self.SetStatusText('Selected File: {}'.format(path))
		# 파일 오픈 다이얼로그 파괴
		dlg.Destroy()

	# 페이지 설정 다이얼로그 이벤트
	def on_pagesetup(self, event):
		dlg = wx.PageSetupDialog(self)
		# 페이지  다이얼로그에서 페이지 설정 후 Ok 버튼을 누르면        
		if dlg.ShowModal() == wx.ID_OK:
			# 페이지 설정 데이타 클래스 구함
			data = dlg.GetPageSetupData()
			# (상단, 왼쪽) 여백을 구함
			tl = data.GetMarginTopLeft()
			# (하단, 오른쪽) 여백을 구함
			br = data.GetMarginBottomRight()
			# 설정한 페이지 설정값을 Frame의 상태바에 출력함
			self.SetStatusText('Margins: {} {}'.format(tl, br))
		# 페이지 설정 다이얼로그 파괴
		dlg.Destroy()

	# 폰트 선택 다이얼로그 이벤트
	def on_choosefont(self, event):
		# 기본 폰트 생성
		default_font = wx.Font(10, wx.FONTFAMILY_SWISS, wx.FONTSTYLE_NORMAL, wx.FONTWEIGHT_NORMAL, False, 'Verdana')
		# 폰트정보 클래스 생성
		data = wx.FontData()
		# 시스템의 플랫폼이 Win32라면
		if sys.platform == 'win32':
			# 폰트 이펙트 표시함.(False: 표시하지 않음)
			data.EnableEffects(True)
		# Windows에서 심볼 글꼴을 선택할 수 없음
		data.SetAllowSymbols(False)
		# 기본 폰트로 초기화
		data.SetInitialFont(default_font)
		# 폰트 크기 지정 범위
		data.SetRange(10, 30)
		# 폰트 다이얼로그 생성
		dlg = wx.FontDialog(self, data)
		# 폰트  다이얼로그에서 폰트 설정 후 Ok 버튼을 누르면
		if dlg.ShowModal() == wx.ID_OK:
			# 설정한 폰트 정보 가져오기
			data = dlg.GetFontData()
			font = data.GetChosenFont() # 글꼴
			color = data.GetColour() # 글자 색상
			print('color =', color, color.Get())
			# 설정한 페이지 설정을 상태바에 출력
			text = 'Face: {}, Size: {}, Color: {}'.format(font.GetFaceName(), font.GetPointSize(),  color.Get())
			self.SetStatusText(text)
		# 폰트 다이얼로그 파괴
		dlg.Destroy()

	# 디렉토리 다이얼로그 이벤트
	def on_opendir(self, event):
		# 디렉토리 다이얼로그 생성
		dlg = wx.DirDialog(self, '폴더 찾아보기', style=wx.DD_DEFAULT_STYLE | wx.DD_NEW_DIR_BUTTON)
		# 디렉토리 다이얼로그에서 Ok 버튼을 누른 경우
		if dlg.ShowModal() == wx.ID_OK:
			dir_path = dlg.GetPath()
			# 선택한 폴더 경로를 상태바에 출력
			self.SetStatusText('Selected directory: {}'.format(dir_path))
		# 디렉토리 다이얼로그 파괴
		dlg.Destroy()

	# 선택 다이얼로그 이벤트
	def on_singlechoice(self, event):
		sins = ['인색(Greed)', '음욕(Lust)', '탐욕(Gluttony)', '교만(Pride)', '나태(Sloth)', '질투(Envy)', '분노(Wrath)'] # 선택할 리스트
		dlg = wx.SingleChoiceDialog(self, '칠죄종', '어느 것을 선택하시나요?', sins, wx.CHOICEDLG_STYLE)
		# 선택 다이얼로그에서 Ok 버튼을 누른 경우
		if dlg.ShowModal() == wx.ID_OK:
			index = dlg.GetSelection()
			select = dlg.GetStringSelection()
			# 선택한 항목을 상태바에 출력
			self.SetStatusText('Single chose: {}. {}'.format(index + 1, select))
		# 선택 다이얼로그 파괴
		dlg.Destroy()

	# 텍스트 입력 다이얼로그 이벤트
	def on_textentry(self, event):
		dlg = wx.TextEntryDialog(self, '텍스트를 입력하세요.', '메모', value='')
		if dlg.ShowModal() == wx.ID_OK:
			text = dlg.GetValue()
			# 입력한 항목을 상태바에 출력
			self.SetStatusText('Entered: {}'.format(text))
		# 텍스트 입력 다이얼로그 파괴
		dlg.Destroy()

	# 재정렬 다이얼로그 이벤트
	def on_rearrange(self, event):
		items = ["사과", "바나나", "체리", "포도", "오렌지"]
		order = [0, 1, 2, 3, 4]
		dlg = wx.RearrangeDialog(self, '과일 목록의 순서를 바꾸거나 숨길 항목을 선택하세요.', '항목 재정렬', items=items, order=order)
		if dlg.ShowModal() == wx.ID_OK:
			new_order = dlg.GetOrder() # 새로운 순서로 정렬된 리스트
			visible_items = [items[i] for i in new_order if i >= 0]
			# 새로 정렬된 리스트를 상태바에 출력
			self.SetStatusText(f'새로 정렬된 리스트: {", ".join(visible_items)}')
		dlg.Destroy()

class MyApp(wx.App):
	def OnInit(self):
		self.locale = wx.Locale(wx.LANGUAGE_KOREAN)
		frame = MyFrame(None, '미리 정의된 다이얼로그(Dialog)')
		frame.Show(True)
		frame.Centre()
		return True

if __name__ == '__main__':
	app = MyApp()
	app.MainLoop()
```

메뉴를 생성하는 구문을 보면 **id** 값을 설정한 다음 이벤트를 처리하는 방식과 **변수**에 메뉴를 지정한 다음 이벤트를 처리하는 방식이 있어요.
저는 주로 변수를 지정하는 방식으로 많이 사용해요. 왜냐하면 **id** 값을 설정할 경우 중복되지 않게 설정해줘야 하는데 코드가 많아지다 보면 이미 사용한 **id**인지 구분하기 어려울 수 있어요.

다이얼로그에서 **style**을 어떻게 설정하는지에 따라 다른 기능을 하는데요. 예를 들어 `wx.FileDialog`의 경우 `wx.FD_OPEN`는 열기 대화상자가 열리지만, `wx.FD_SAVE`는 저장 대화상자가 열리게 되요.
자세한 내용은 [공식 문서](https://docs.wxpython.org/)를 참고하시면 되요.

> **Note**:  
> `wx.MessageDialog`에서 `style=wx.ICON_QUESTION`로 설정할 경우 **물음표 아이콘**이 나와야 하는데 안 나오는 문제가 있어요.  
> 그래서 찾아보니 윈도우에서 지원을 안 한다고 하는데 또 이상한게 **ctypes** 패키지에서는 정상적으로 나오더라고요.
> ```python
> import ctypes
> ctypes.windll.user32.MessageBoxW(32, '프로그램을 종료할까요?', '알림')
> ```
{: .notice--warning}

> **wx.MessageBox**  
> 간단한 알림창을 띄우기 위해 `wx.MessageDialog`를 사용하려면 `.ShowModal()`과 `.Destroy()` 메서드를 호출해줘야 해서 번거로울 수 있는데요.  
> 그래서 많이 사용하는게 `wx.MessageBox`에요.  
> ```python
> answer = wx.MessageBox('프로그램을 종료할까요?', '알림', wx.YES_NO | wx.CANCEL)
> if answer == wx.YES:
> 	print('answer:', answer)
> ```
> 차이점이라고 하면 반환 값이 `wx.ID_OK`가 아니라 `wx.OK`로 되니 참고하세요.
{: .notice--primary}

## 커스텀 다이얼로그

지금까지 **wxPython**에서 미리 정의되어 있는 **다이얼로그(Dialog)**를 만들어 보았는데요.
이번에는 커스텀 다이얼로그를 만드는 방법에 대해 알려드릴게요.

```python
# -*- coding: utf-8 -*-
# custom_dialog.py

import wx

class MyApp(wx.App):
	def OnInit(self):
		# 메모 다이얼로그
		dlg = wx.Dialog(None, title='메모', size=(400, 400))
		static_title = wx.StaticText(dlg, label='제목')
		tc_title = wx.TextCtrl(dlg)
		static_memo = wx.StaticText(dlg, label='내용')
		tc_memo = wx.TextCtrl(dlg, style=wx.TE_MULTILINE)
		btn_ok = wx.Button(dlg, wx.ID_OK, '작성하기')

		# 사이저 할당
		vbox = wx.BoxSizer(wx.VERTICAL)
		vbox.AddMany([ (static_title, 0, wx.ALL, 5),
			(tc_title, 0, wx.EXPAND|wx.ALL, 5),
			(static_memo, 0, wx.ALL, 5),
			(tc_memo, 1, wx.EXPAND|wx.ALL, 5),
			(btn_ok, 0, wx.ALL|wx.ALIGN_CENTER, 5) ])
		dlg.SetSizer(vbox)

		if dlg.ShowModal() == wx.ID_OK:
			title = tc_title.GetValue()
			print('제목:', title)
			memo = tc_memo.GetValue()
			print('메모:', memo)
		dlg.Destroy()
		return True

if __name__ == '__main__':
	app = MyApp()
	app.MainLoop()
```

위 코드는 간단하게 **다이얼로그(Dialog)**를 만들 때 사용하는 방식이고요. 보통은 클레스에 `wx.Dialog`를 부모 클레스로 선언해서 작성하면 됩니다.
이번에는 위 코드를 클래스로 만들어 볼게요.

```python
# -*- coding: utf-8 -*-
# memo_dialog.py

import wx

# 메모 다이얼로그
class MemoDialog(wx.Dialog):
	def __init__(self, parent, title: str, size: tuple):
		super().__init__(parent, title=title, size=size)

		static_title = wx.StaticText(self, label='제목:')
		self.tc_title = wx.TextCtrl(self)
		static_memo = wx.StaticText(self, label='내용:')
		self.tc_memo = wx.TextCtrl(self, style=wx.TE_MULTILINE)
		btn_ok = wx.Button(self, wx.ID_OK, '작성하기')

		# 사이저 할당
		vbox = wx.BoxSizer(wx.VERTICAL)
		vbox.AddMany([ (static_title, 0, wx.ALL, 5),
			(self.tc_title, 0, wx.EXPAND, 5),
			(static_memo, 0, wx.ALL, 5),
			(self.tc_memo, 1, wx.EXPAND, 5),
			(btn_ok, 0, wx.ALL|wx.ALIGN_CENTER, 5) ])
		self.SetSizer(vbox)

class MyApp(wx.App):
	def OnInit(self):
		dlg = MemoDialog(None, '메모', (400, 400))
		if dlg.ShowModal() == wx.ID_OK:
			title = dlg.tc_title.GetValue()
			print('제목:', title)
			memo = dlg.tc_memo.GetValue()
			print('메모:', memo)
		dlg.Destroy()
		return True

if __name__ == '__main__':
	app = MyApp()
	app.MainLoop()
```
